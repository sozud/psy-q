POL3

三角形ポリゴン

構造 

typedef struct {
		short sxy[3][2];
		short sz[3][2];
		short uv[3][2];
		short rgb[3][3];
		short code;
} POL3;
メンバ 
sxy		スクリーン座標のxy成分
sz		スクリーン座標のz成分
uv		テクスチャ座標
rgb		頂点の色情報
code		ポリゴンコード
