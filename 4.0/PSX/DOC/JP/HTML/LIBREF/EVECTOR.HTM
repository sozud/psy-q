EVECTOR

クリップベクトル情報

構造 

typedef struct {
		SVECTOR v;
		VECTOR sxyz;
		DVECTOR sxy;
		CVECTOR rgb;
		short txuv,pad;
		long chx,chy;
} EVECTOR;
メンバ 
v		ローカルオブジェクトの３次元頂点
sxyz		スクリーンの３次元頂点
sxy		スクリーンの２次元頂点
rgb		頂点の色成分
txuv,pad	テクスチャマッピングデータ
chx,chy		クリップ領域データ
