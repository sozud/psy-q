POL4

四角形ポリゴン

構造 

typedef struct {
		short sxy[4][2];
		short sz[4][2];
		short uv[4][2];
		short rgb[4][3];
		short code;
} POL4;
メンバ 
sxy		スクリーン座標のxy成分
sz		スクリーン座標のz成分
uv		テクスチャ座標
rgb		頂点の色情報
code		ポリゴンコード
