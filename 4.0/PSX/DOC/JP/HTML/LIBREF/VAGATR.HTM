VagAtr

波形ヘッダ

構造 

struct VagAtr {
		unsigned char prior;
		unsigned char mode;
		unsigned char vol;
		unsigned char pan;
		unsigned char center;
		unsigned char shift;
		unsigned char min;
		unsigned char max;
		unsigned char vibW;
		unsigned char vibT;
		unsigned char porW;
		unsigned char porT;
		unsigned char pbmin;
		unsigned char pbmax;
		unsigned char reserved1;
		unsigned short reserved2;
		unsigned short adsr1;
		unsigned short adsr2;
		short prog;
		short vag;
		short reserved [4];
};
メンバ 
prior		優先度（0～15）
mode		音源モード
vol		ボリューム（0～127, 0:min, 127:max）
pan		パンポット (0～127, 0:left,63:center,127:right)
center		センターノート (0～127)
shift		ピッチ補整（0～99, cent 単位）
min		最小ノートリミット
max		最大ノートリミット
vibW		ビブラート幅（0 ～ 127, 127 で一オクターブ）
vibT		ビブラートの１周期の時間（tick 単位）
porW		ポルタメント幅
porT		ポルタメント持続時間（tick 単位）
pbmin		最小ピッチベンドリミット
pbmax		最大ピッチベンドリミット
reseaved1	システム予約
reseaved2	システム予約
adsr1		ADSR 設定値 (その１)
adsr2		ADSR 設定値 (その２)
prog		この VAG 属性が含まれる親プログラム
vag		この VAG 属性が利用する VAG の ID 番号
reserved [0..3]	システム予約
