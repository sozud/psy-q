DIVPOLYGON4

４角形用分割バッファ　

構造 

typedef struct {
		u_long ndiv;
		u_long pih,piv;
		u_short clut,tpage;
		CVECTOR rgbc;
		u_long *ot;
		RVECTOR r0,r1,r2,r3;
		CRVECTOR4 cr[5];
} DIVPOLYGON4;
メンバ 
ndiv		分割数
pih,piv		クリップエリア指定（表示画面の解像度）
clut		CLUT
tpage		テクスチャページ
rgbc		コード + 色成分
ot		OTへのポインタ
r0,r1,r2,r3	分割頂点ベクトル情報
cr		４角形用再帰ベクトル情報
